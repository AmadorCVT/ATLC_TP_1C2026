/* Programa integral para ejercitar las capacidades aceptadas del DSL. */

// DFA determinista: paridad de cantidad de ceros.
automaton ParidadCeros : DFA {
    alphabet = {0, 1}
    states = {q0, q1}
    start = q0
    accept = {q0}
    transitions {
        q0 -> 0 : q1
        q0 -> 1 : q0
        q1 -> 0 : q0
        q1 -> 1 : q1
    }
};

String par = "0011";
String impar = "0010";
String vacia = "";

print ParidadCeros;
show transitions of ParidadCeros;
show table of ParidadCeros;
test ParidadCeros with par;
test ParidadCeros with impar;
test ParidadCeros with "1111";

for palabra in {"00", "0010", "1010"} {
    test ParidadCeros with palabra;
}

convert ParidadCeros to NFA as ParidadCerosNFA;
test ParidadCerosNFA with "00";

// NFA con multiples destinos y conversion por construccion de subconjuntos.
automaton TieneAB : NFA {
    alphabet = {a, b}
    states = {s0, s1, s2}
    start = s0
    accept = {s2}
    transitions {
        s0 -> a : {s0, s1}
        s0 -> b : {s0}
        s1 -> b : {s2}
        s2 -> a : {s2}
        s2 -> b : {s2}
    }
};

show table of TieneAB;
test TieneAB with "bbbbab";
test TieneAB with "aaaa";
convert TieneAB to DFA as TieneABDFA;
show table of TieneABDFA;
test TieneABDFA with "baba";
test TieneABDFA with "bbb";

// LNFA con lambda-closure y conversiones soportadas a NFA y DFA.
automaton LambdaAB : LNFA {
    alphabet = {a, b}
    states = {l0, l1, l2}
    start = l0
    accept = {l2}
    transitions {
        l0 -> lambda : {l1}
        l1 -> a : {l1}
        l1 -> b : {l2}
        l2 -> a : {l2}
        l2 -> b : {l2}
    }
};

show closure(l0) in LambdaAB;
show table of LambdaAB;
test LambdaAB with "aaab";
test LambdaAB with vacia;
convert LambdaAB to NFA as LambdaABNFA;
convert LambdaAB to DFA as LambdaABDFA;
show transitions of LambdaABNFA;
show table of LambdaABDFA;
test LambdaABNFA with "ab";
test LambdaABDFA with "aaaa";

// LNFA que acepta la cadena vacia por lambda-closure.
automaton LambdaVacia : LNFA {
    alphabet = {a}
    states = {e0, e1}
    start = e0
    accept = {e1}
    transitions {
        e0 -> lambda : {e1}
        e1 -> a : {e1}
    }
};

show closure(e0) in LambdaVacia;
test LambdaVacia with vacia;
test LambdaVacia with "aaa";

// Update valido: agrega estado, cambia aceptacion y completa transiciones.
automaton Editable : DFA {
    alphabet = {x}
    states = {u0}
    start = u0
    accept = {u0}
    transitions {
        u0 -> x : u0
    }
};

Editable {
    states = {u1}
    accept = {u1}
    transitions {
        u0 -> x : u1
        u1 -> x : u1
    }
};

show transitions of Editable;
test Editable with vacia;
test Editable with "x";
